236 void Layout::processUnhandledTextureSlots()
237 {
238 auto it = m_unhandledTextureSlots.begin();
239 while (it != m_unhandledTextureSlots.end())
240 {
241 FCT::ShaderStages usage;
242
243 if (!m_hasVertexLayout && !m_hasPixelLayout)
244 {
245 ++it;
246 continue;
247 }
248 else if (m_hasVertexLayout && !m_hasPixelLayout)
249 {
250 usage = FCT::ShaderStage::Vertex | FCT::ShaderStage::Fragment;
251 }
252 else if (!m_hasVertexLayout && m_hasPixelLayout)
253 {
254 usage = FCT::ShaderStage::Vertex | FCT::ShaderStage::Fragment;
255 }
256 else
257 {
258 usage = FCT::ShaderStage::Fragment;
259 }
260 m_textureNames[it->name] = nullptr;
261 m_textureNames[it->name] = m_textureNames.find(it->name)->first.c_str();
262 FCT::TextureElement element(FCT::TextureType::Texture2D, m_textureNames[it->name], usage, FCT::UpdateFrequency::PerFrame);
263 addTextureSlot(element);
264
265 it = m_unhandledTextureSlots.erase(it);
266 }
267 }

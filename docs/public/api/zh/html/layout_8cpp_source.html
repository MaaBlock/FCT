368 void Layout::processUnhandledTextureSlots()
369 {
370 auto it = m_unhandledTextureSlots.begin();
371 while (it != m_unhandledTextureSlots.end())
372 {
373 FCT::ShaderStages usage;
374
375 if (!m_hasVertexLayout && !m_hasPixelLayout)
376 {
377 ++it;
378 continue;
379 }
380 else if (m_hasVertexLayout && !m_hasPixelLayout)
381 {
382 usage = FCT::ShaderStage::Vertex | FCT::ShaderStage::Fragment;
383 }
384 else if (!m_hasVertexLayout && m_hasPixelLayout)
385 {
386 usage = FCT::ShaderStage::Vertex | FCT::ShaderStage::Fragment;
387 }
388 else
389 {
390 usage = FCT::ShaderStage::Fragment;
391 }
392 m_textureNames[it->name] = nullptr;
393 m_textureNames[it->name] = m_textureNames.find(it->name)->first.c_str();
394 FCT::TextureElement element(FCT::TextureType::Texture2D, m_textureNames[it->name], usage, FCT::UpdateFrequency::PerFrame);
395 addTextureSlot(element);
396
397 it = m_unhandledTextureSlots.erase(it);
398 }
399 }

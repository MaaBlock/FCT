\begin{DoxyItemize}
\item X、\+Y坐标：从\mbox{[}-\/1,1\mbox{]}映射到实际屏幕像素范围（如0-\/1919, 0-\/1079）
\item Z坐标：从\mbox{[}-\/1,1\mbox{]}映射到深度缓冲区范围（通常是\mbox{[}0,1\mbox{]}）
\end{DoxyItemize}\hypertarget{md_docs_2zh_2md_2how__3d__object__rendered_autotoc_md17}{}\doxysubsection{\texorpdfstring{5. 光栅化}{5. 光栅化}}\label{md_docs_2zh_2md_2how__3d__object__rendered_autotoc_md17}

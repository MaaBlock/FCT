516 void partitionPassGroups()
517 {
518 m_tokenGraph.visitDFSFromRoots([this](RenderGraphNode node)
519 {
520 m_compiledPasses[node.name].name = node.name;
521 auto getResourceGroup = [this](PassDesc& pass) -> std::string
522 {
523 std::string resourceGroup;
524 if (pass.m_targetsCount > 0)
525 {
526 resourceGroup = m_bufferUnions.find(pass.targets[0].name);
527 return resourceGroup;
528 }
529 if (pass.depthStencil.name.size())
530 {
531 resourceGroup = m_bufferUnions.find(pass.depthStencil.name);
532 return resourceGroup;
533 }
534 return "error?";
535 };
536 m_compiledPasses[node.name].resourceGroup = getResourceGroup(m_originPasses[node.name]);
537 });
538
539 m_tokenGraph.visitDFSFromRoots([this](RenderGraphNode node)
540 {
541 auto inputPasses = m_tokenGraph.getActivePredecessors(
542 node);
543 if (isMutilInputNode(node.name))
544 {
545 m_passUnions.unite(node.name,node.name);

547 }
548 for (auto& inputPass : inputPasses)

551 {
552 m_passUnions.unite(node.name,node.name);
553 return;
554 }

580 void buildPassGroupGraph()
581 {
582 auto unions = m_passUnions.getGroups();
583 for (auto& [groupRoot, passes] : unions)
584 {
585 CompiledGroup compiledGroup;
586 compiledGroup.passes = passes;
592 std::set<std::string> groupInputBuffers;
593 std::set<std::string> groupOutputBuffers;
594 std::set<std::string> internalBuffers;
595
596 for (const auto& passName : passes)
597 {
598 const auto& pass = m_originPasses[passName];
599
600 for (size_t i = 0; i < pass.m_textureCount; ++i)
601 {
602 const std::string& bufferName = pass.textures[i].name;
603
604 bool isInternal = false;
605 for (const auto& otherPassName : passes)
606 {
607 if (otherPassName == passName) continue;
608 const auto& otherPass = m_originPasses[otherPassName];
609
610 for (size_t j = 0; j < otherPass.m_targetsCount; ++j)
611 {
612 if (otherPass.targets[j].name == bufferName)
613 {
614 isInternal = true;
615 break;
616 }
617 }
618
619 if (!isInternal && otherPass.depthStencil.name == bufferName)
620 {
621 isInternal = true;
622 }
623
624 if (isInternal) break;
625 }
626
627 if (isInternal)
628 {
629 internalBuffers.insert(bufferName);
630 }
631 else
632 {
633 groupInputBuffers.insert(bufferName);
634 }
635 }
636
637 for (size_t i = 0; i < pass.m_targetsCount; ++i)
638 {
639 const std::string& bufferName = pass.targets[i].name;
640
641 bool isInternal = false;
642 for (const auto& otherPassName : passes)
643 {
644 if (otherPassName == passName) continue;
645 const auto& otherPass = m_originPasses[otherPassName];
646
647 for (size_t j = 0; j < otherPass.m_textureCount; ++j)
648 {
649 if (otherPass.textures[j].name == bufferName)
650 {
651 isInternal = true;
652 break;
653 }
654 }
655
656 if (isInternal) break;
657 }
658
659 if (isInternal)
660 {
661 internalBuffers.insert(bufferName);
662 }
663 else
664 {
665 groupOutputBuffers.insert(bufferName);
666 }
667 }
668
669 if (!pass.depthStencil.name.empty())
670 {
671 const std::string& bufferName = pass.depthStencil.name;
672
673 bool isInternal = false;
674 for (const auto& otherPassName : passes)
675 {
676 if (otherPassName == passName) continue;
677 const auto& otherPass = m_originPasses[otherPassName];
678
679 for (size_t j = 0; j < otherPass.m_textureCount; ++j)
680 {
681 if (otherPass.textures[j].name == bufferName)
682 {
683 isInternal = true;
684 break;
685 }
686 }
687
688 if (isInternal) break;
689 }
690
691 if (isInternal)
692 {
693 internalBuffers.insert(bufferName);
694 }
695 else
696 {
697 groupOutputBuffers.insert(bufferName);
698 }
699 }
700 }
701
702 compiledGroup.inputs = groupInputBuffers;
703 compiledGroup.internals = internalBuffers;
704 compiledGroup.outputs = groupOutputBuffers;
705
706 m_compiledGroups[groupRoot] = compiledGroup;
707 PassGroupRenderGraphNode groupNode{groupRoot, PassGroupRenderGraphNodeType::group};
708
709 std::vector<PassGroupRenderGraphNode> predecessors;
710 for (const auto& inputBuffer : compiledGroup.inputs) {
711 predecessors.emplace_back(PassGroupRenderGraphNode{inputBuffer, PassGroupRenderGraphNodeType::image});
712 }
713
714 std::vector<PassGroupRenderGraphNode> successors;
715 for (const auto& outputBuffer : compiledGroup.outputs) {
716 successors.emplace_back(PassGroupRenderGraphNode{outputBuffer, PassGroupRenderGraphNodeType::image});
717 }
718
719 m_groupGraph[groupNode] = { groupNode, predecessors, successors};

208 class RenderGraph {
209 private:
210 Context* m_ctx;
211 Device* m_device;
212 std::unordered_map<std::string_view,PassDesc> m_originPasses;
213 std::unordered_map<std::string_view,ImageDesc> m_originImages;
214 void collectImages()
215 {
216 for (auto& [name, pass] : m_originPasses)
217 {
218 for (size_t i = 0; i < pass.m_textureCount; ++i)
219 {
220 auto& texture = pass.textures[i];
221 if (texture.format != Format::UNDEFINED)
222 {
223 m_originImages[texture.name].format = texture.format;
224 }
225 if (texture.size)
226 {
227 m_originImages[texture.name].size = texture.size;
228 }
229 if (texture.samples!= Samples::sample_undefined)
230 {
231 m_originImages[texture.name].samples = texture.samples;
232 }
233 m_originImages[texture.name].usages |= ImageUsage::Texture;
234 }
235 for (size_t i = 0; i < pass.m_targetsCount; ++i)
236 {
237 auto& target = pass.targets[i];
238 if (target.wnd)
239 {
240 m_originImages[target.name].wnd = target.wnd;
241 continue;
242 }
243 if (target.size)
244 {
245 m_originImages[target.name].size = target.size;
246 }
247 if (target.format!= Format::UNDEFINED)
248 {
249 m_originImages[target.name].format = target.format;
250 }
251 if (target.samples!= Samples::sample_undefined)
252 {
253 m_originImages[target.name].samples = target.samples;
254 }
255 m_originImages[target.name].usages |= ImageUsage::RenderTarget;
256 }
257 if (pass.depthStencil.name.data())
258 {
259 auto& depthStencil = pass.depthStencil;
260 if (depthStencil.format!= Format::UNDEFINED)
261 {
262 m_originImages[depthStencil.name].format = depthStencil.format;
263 }
264 if (depthStencil.size)
265 {
266 m_originImages[depthStencil.name].size = depthStencil.size;
267 }
268 if (depthStencil.samples != Samples::sample_undefined)
269 {
270 m_originImages[depthStencil.name].samples = depthStencil.samples;
271 }
272 m_originImages[depthStencil.name].usages |= ImageUsage::DepthStencil;
273 }
274 }
275 }
276 public:
277 RenderGraph(Context* ctx);
278 ~RenderGraph();
279 template<typename... Args>
280 void addPass(std::string_view name,Args&&... args) {
281 PassDesc pass(name);
282 pass.processArgs(std::forward<Args>(args)...);
283 m_originPasses[name] = pass;
284 }
285 void complie()
286 {
287 collectImages();
288 }
289 };

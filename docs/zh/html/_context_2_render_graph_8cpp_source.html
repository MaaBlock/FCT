192 void RenderGraph::resolveTextureSizes()
193 {
194 for (auto& [passName, pass] : m_passNodes) {
195 std::vector<RenderGraphImageNode*> referenceNodes;
196
197 for (auto* targetEdge : pass.getTargetOutgoingEdges()) {
198 auto imageNodeIt = m_imageNodes.find(targetEdge->toImage);
199 if (imageNodeIt != m_imageNodes.end()) {
200 referenceNodes.push_back(imageNodeIt->second.get());
201 }
202 }
203
204 for (auto* depthStencilEdge : pass.getDepthStencilOutgoingEdges()) {
205 auto imageNodeIt = m_imageNodes.find(depthStencilEdge->toImage);
206 if (imageNodeIt != m_imageNodes.end()) {
207 referenceNodes.push_back(imageNodeIt->second.get());
208 }
209 }
210
211 if (referenceNodes.empty()) {
212 continue;
213 }
214
215 RenderGraphImageNode* referenceNode = referenceNodes[0];
216
217 for (auto* textureEdge : pass.getTextureIncomingEdges()) {
218 auto imageNodeIt = m_imageNodes.find(textureEdge->fromImage);
219 if (imageNodeIt == m_imageNodes.end()) {
220 continue;
221 }
222
223 RenderGraphImageNode* textureNode = imageNodeIt->second.get();
224
225 if (textureEdge->size.type == TextureSize::Type::Undefined) {
226 if (!textureNode->isSizeDetermined()) {
227 textureNode->unite(referenceNode);
228 }
229 }
230 else if (textureEdge->size.type == TextureSize::Type::Relative) {
231 FractionScale2D scale(textureEdge->size.relativeWidth, textureEdge->size.relativeHeight);
232 textureNode->unite(referenceNode, scale);

313 void RenderGraph::allocateResources()
314 {
315 for (auto& [name, imageNode] : m_imageNodes) {
316 imageNode->fillDefaultData();
317 }
318 auto* resourceManager = m_ctx->getModule<ResourceManager>();
319 if (!resourceManager) {
320 throw std::runtime_error("ResourceManager not available");
321 }
322
323 std::set<const SizeNode*> processedRoots;
324
325 for (auto& [name, imageNode] : m_imageNodes) {
326 auto* bufferNode = dynamic_cast<RenderGraphBufferNode*>(imageNode.get());
327 if (!bufferNode || !bufferNode->isFilled()) {
328 continue;
329 }
330
331 const SizeNode* root = bufferNode->getRoot();
332
333 if (root == bufferNode && processedRoots.find(root) == processedRoots.end()) {
334 processedRoots.insert(root);
335
336 uint32_t width, height;
337 bool hasFixedSize = bufferNode->getComputedSize(width, height);
338
339 if (hasFixedSize) {
340 TargetDesc targetDesc;
341 targetDesc.width = static_cast<int>(width);
342 targetDesc.height = static_cast<int>(height);
343 targetDesc.samples = bufferNode->getSamples();
344 targetDesc.format = bufferNode->getFormat();
345 targetDesc.usage = bufferNode->getUsage();
346
347 Image* allocatedImage = resourceManager->allocateTarget(name, targetDesc);
348 bufferNode->setAllocatedImage(allocatedImage); // 调用setAllocatedImage保存图像
349 m_allocatedImages[name] = allocatedImage; // 存储分配的图像
350 } else {
351 throw std::runtime_error("Root node has no fixed size: " + name);
352 }
353 }
354 }
355
356 for (auto& [name, imageNode] : m_imageNodes) {
357 auto* bufferNode = dynamic_cast<RenderGraphBufferNode*>(imageNode.get());
358 if (!bufferNode || !bufferNode->isFilled()) {
359 continue;
360 }
361
362 const SizeNode* root = bufferNode->getRoot();
363
364 if (root != bufferNode) {
365 ImageDesc imageDesc;
366 imageDesc.format = bufferNode->getFormat();
367 imageDesc.samples = bufferNode->getSamples();
368 imageDesc.usage = bufferNode->getUsage();
369
370 Image* allocatedImage = nullptr;
371
372 std::string dependencyName;
373 for (auto& [otherName, otherNode] : m_imageNodes) {
374 if (otherNode->getRoot() == root && otherNode->getRoot() == otherNode.get()) {
375 dependencyName = otherName;
376 break;
377 }
378 }
379
380 if (!dependencyName.empty()) {
381 auto* windowTargetNode = dynamic_cast<RenderGraphWindowTargetNode*>(
382 m_imageNodes[dependencyName].get());
383 auto* windowDepthNode = dynamic_cast<RenderGraphWindowDepthStencilNode*>(
384 m_imageNodes[dependencyName].get());
385
386 if (windowTargetNode && windowTargetNode->isValidWindowTarget()) {
387 allocatedImage = resourceManager->allocateImage(
388 name, windowTargetNode->getWindow(), imageDesc);
389 } else if (windowDepthNode && windowDepthNode->isValidWindowDepthStencil()) {
390 allocatedImage = resourceManager->allocateImage(
391 name, windowDepthNode->getWindow(), imageDesc);
392 } else {
393 allocatedImage = resourceManager->allocateImage(
394 name, dependencyName, imageDesc);
395 }
396
397 bufferNode->setAllocatedImage(allocatedImage);
398 m_allocatedImages[name] = allocatedImage;
399 } else {
400 throw std::runtime_error("Cannot determine dependency for image: " + name);
401 }
402 }
403 }

441 void RenderGraph::analyzePassGroupDependencies(
442 const std::string& groupLeader,
443 const std::vector<std::string>& groupMembers,
444 std::map<std::string, std::set<std::string>>& dependencies
445) const {
446 if (dependencies.find(groupLeader) == dependencies.end()) {
447 dependencies[groupLeader] = std::set<std::string>();
448 }
449
450 for (const std::string& passName : groupMembers) {
451 auto passNodeIt = m_passNodes.find(passName);
452 if (passNodeIt == m_passNodes.end()) continue;
453
454 const auto& passNode = passNodeIt->second;
455
456 for (const auto* textureEdge : passNode.getTextureIncomingEdges()) {
457 auto imageNodeIt = m_imageNodes.find(textureEdge->fromImage);
458 if (imageNodeIt == m_imageNodes.end()) continue;
459
460 const auto& imageNode = imageNodeIt->second;
461
462 for (const auto* targetEdge : imageNode->getTargetIncomingEdges()) {
463 std::string producerPass = targetEdge->fromPass;
464
465 auto allGroups = m_passesUnions.getGroups();
466 for (const auto& [otherGroupLeader, otherGroupMembers] : allGroups) {
467 if (otherGroupLeader != groupLeader &&
468 std::find(otherGroupMembers.begin(), otherGroupMembers.end(), producerPass) != otherGroupMembers.end()) {
469 dependencies[groupLeader].insert(otherGroupLeader);
470 break;
471 }
472 }
473 }
474
475 for (const auto* depthEdge : imageNode->getDepthStencilIncomingEdges()) {
476 std::string producerPass = depthEdge->fromPass;
477
478 auto allGroups = m_passesUnions.getGroups();
479 for (const auto& [otherGroupLeader, otherGroupMembers] : allGroups) {
480 if (otherGroupLeader != groupLeader &&
481 std::find(otherGroupMembers.begin(), otherGroupMembers.end(), producerPass) != otherGroupMembers.end()) {
482 dependencies[groupLeader].insert(otherGroupLeader);
483 break;
484 }
485 }
486 }
487 }

491 std::vector<std::string> RenderGraph::topologicalSort(
492 const std::map<std::string, std::set<std::string>>& dependencies
493 ) const {
494 std::vector<std::string> result;
495 std::map<std::string, int> inDegree;
496 std::queue<std::string> queue;
497
498 for (const auto& [node, deps] : dependencies) {
499 inDegree[node] = 0;
500 }
501
502 for (const auto& [node, deps] : dependencies) {
503 for (const std::string& dep : deps) {
504 if (inDegree.find(dep) != inDegree.end()) {
505 inDegree[node]++;
506 }
507 }
508 }
509
510 for (const auto& [node, degree] : inDegree) {
511 if (degree == 0) {
512 queue.push(node);
513 }
514 }
515
516 while (!queue.empty()) {
517 std::string current = queue.front();
518 queue.pop();
519 result.push_back(current);
520
521 for (const auto& [node, deps] : dependencies) {
522 if (deps.find(current) != deps.end()) {
523 inDegree[node]--;
524 if (inDegree[node] == 0) {
525 queue.push(node);
526 }
527 }
528 }
529 }
530
531 if (result.size() != dependencies.size()) {
532 throw std::runtime_error("Circular dependency detected in PassGroups");
533 }
534
535 return result;
536 }

557 void RenderGraph::createPassGroups() {
558 auto passGroups = m_passesUnions.getGroups();
559
560 for (const auto& [groupLeader, groupMembers] : passGroups) {
561 RHI::PassGroup* passGroup = m_ctx->createResource<RHI::PassGroup>();
562
563 std::vector<std::string> orderedPassNames;
564
565 for (const std::string& passName : groupMembers) {
566 auto passIt = m_allocatedPasses.find(passName);
567 if (passIt == m_allocatedPasses.end()) continue;
568
569 RHI::Pass* currentPass = passIt->second;
570 std::vector<RHI::PassNeighbors> predecessors;
571 std::vector<RHI::PassNeighbors> successors;
572
573 auto passNodeIt = m_passNodes.find(passName);
574 if (passNodeIt != m_passNodes.end()) {
575 const auto& passNode = passNodeIt->second;
576
577 bool outputsToWindow = false;
578 //判断 target和depth是否窗口
579 for (const auto* targetEdge : passNode.getTargetOutgoingEdges()) {
580 auto imageNodeIt = m_imageNodes.find(targetEdge->toImage);
581 if (imageNodeIt != m_imageNodes.end()) {
582 auto* windowTargetNode = dynamic_cast<RenderGraphWindowTargetNode*>(
583 imageNodeIt->second.get());
584 if (windowTargetNode && windowTargetNode->isValidWindowTarget()) {
585 outputsToWindow = true;
586 break;
587 }
588 }
589 }
590
591 if (!outputsToWindow) {
592 for (const auto* depthEdge : passNode.getDepthStencilOutgoingEdges()) {
593 auto imageNodeIt = m_imageNodes.find(depthEdge->toImage);
594 if (imageNodeIt != m_imageNodes.end()) {
595 auto* windowDepthNode = dynamic_cast<RenderGraphWindowDepthStencilNode*>(
596 imageNodeIt->second.get());
597 if (windowDepthNode && windowDepthNode->isValidWindowDepthStencil()) {
598 outputsToWindow = true;
599 break;
600 }
601 }
602 }
603 }
604
605 if (outputsToWindow) {
606 successors.emplace_back(
607 RHI::Pass::present
608 );
609 }
610
611 for (const auto* textureEdge : passNode.getTextureIncomingEdges()) {
612 auto imageNodeIt = m_imageNodes.find(textureEdge->fromImage);
613 if (imageNodeIt != m_imageNodes.end()) {
614 const auto& imageNode = imageNodeIt->second;
615
616 // 检查Target边
617 for (const auto* targetEdge : imageNode->getTargetIncomingEdges()) {
618 if (std::find(groupMembers.begin(), groupMembers.end(), targetEdge->fromPass) != groupMembers.end()) {
619 auto predecessorPassIt = m_allocatedPasses.find(targetEdge->fromPass);
620 if (predecessorPassIt != m_allocatedPasses.end()) {
621 predecessors.emplace_back(
622 predecessorPassIt->second,
623 PipelineStage::colorAttachmentOutput,
624 AccessFlag::colorAttachmentWrite,
625 convertShaderStageToPipelineStage(textureEdge->stage),
626 AccessFlag::shaderRead
627 );
628 }
629 }
630 }
631
632 // 检查DepthStencil边
633 for (const auto* depthEdge : imageNode->getDepthStencilIncomingEdges()) {
634 if (std::find(groupMembers.begin(), groupMembers.end(), depthEdge->fromPass) != groupMembers.end()) {
635 auto predecessorPassIt = m_allocatedPasses.find(depthEdge->fromPass);
636 if (predecessorPassIt != m_allocatedPasses.end()) {
637 predecessors.emplace_back(
638 predecessorPassIt->second,
639 PipelineStage::lateFragmentTests,
640 AccessFlag::depthStencilAttachmentWrite,
641 convertShaderStageToPipelineStage(textureEdge->stage),
642 AccessFlag::shaderRead
643 );
644 }
645 }
646 }
647 }
648 }
649 }
650
651 orderedPassNames.push_back(passName);
652
653 passGroup->addPass({
654 currentPass,
655 predecessors,
656 successors
657 });
658 }
659
660 m_passGroupOrders[groupLeader] = std::move(orderedPassNames);
661
662 passGroup->create();
663
664 m_allocatedPassGroups[groupLeader] = passGroup;
665 }

667 void RenderGraph::submitPassGroup(RHI::CommandBuffer* cmdBuffer,
668 const std::string& groupLeader) {
669 auto passGroup = m_allocatedPassGroups.at(groupLeader);
670 auto orderIt = m_passGroupOrders.find(groupLeader);
671 if (orderIt == m_passGroupOrders.end()) {
672 return;
673 }
674
675 const std::vector<std::string>& orderedPassNames = orderIt->second;
676
677 passGroup->beginSubmit(cmdBuffer);
678
679 for (size_t i = 0; i < orderedPassNames.size(); ++i) {
680 const std::string& passName = orderedPassNames[i];
681
682 PassSubmitEvent env;
683 env.passName = passName;
684 env.cmdBuf = cmdBuffer;
685 trigger(env);
686
687 if (i < orderedPassNames.size() - 1) {
688 passGroup->nextPass(cmdBuffer);
689 }
690 }
691
692 passGroup->endSubmit(cmdBuffer);

721 std::vector<RenderGraph::BarrierInfo> RenderGraph::checkBarriersBeforePassGroup(
722 const std::string& groupLeader,
723 const std::vector<std::string>& groupMembers,
724 const std::unordered_map<Image*, ImageState>& imageStates) {
725
726 std::vector<BarrierInfo> barriers;
727 std::set<Image*> processedImages;
728
729 for (const std::string& passName : groupMembers) {
730 auto passIt = m_passNodes.find(passName);
731 if (passIt == m_passNodes.end()) continue;
732
733 const auto& passNode = passIt->second;
734
735 for (const auto* textureEdge : passNode.getTextureIncomingEdges()) {
736 auto imageNodeIt = m_imageNodes.find(textureEdge->fromImage);
737 if (imageNodeIt == m_imageNodes.end()) continue;
738
739 Image* image = imageNodeIt->second->getImage();
740 if (!image || processedImages.count(image)) continue;
741
742 auto stateIt = imageStates.find(image);
743 if (stateIt == imageStates.end()) continue;
744
745 const ImageState& currentState = stateIt->second;
746
747 PipelineStage requiredStage = convertShaderStageToPipelineStage(textureEdge->stage);
748 AccessFlag requiredAccess = AccessFlag::shaderRead;
749 ImageLayout requiredLayout = ImageLayout::shaderReadOnlyOptimal;
750
751 bool needsBarrier = false;
752
753 if (currentState.currentLayout != requiredLayout &&
754 currentState.currentLayout != ImageLayout::undefined) {
755 needsBarrier = true;
756 }
757
758 if (!currentState.lastWriterGroup.empty() &&
759 currentState.lastWriterGroup != groupLeader &&
760 (currentState.lastAccess & (AccessFlag::colorAttachmentWrite | AccessFlag::depthStencilAttachmentWrite))) {
761 needsBarrier = true;
762 }
763
764 if (needsBarrier) {
765 BarrierInfo barrier;
766 barrier.image = image;
767 barrier.oldLayout = currentState.currentLayout;
768 barrier.newLayout = requiredLayout;
769 barrier.srcStage = currentState.lastStage;
770 barrier.dstStage = requiredStage;
771 barrier.srcAccess = currentState.lastAccess;
772 barrier.dstAccess = requiredAccess;
773
774 if (currentState.currentLayout == ImageLayout::depthAttachmentOptimal ||
775 currentState.currentLayout == ImageLayout::depthStencilAttachmentOptimal) {
776 barrier.aspect = ImageAspect::depth;
777 } else {
778 barrier.aspect = ImageAspect::color;
779 }
780
781 barriers.push_back(barrier);
782 processedImages.insert(image);
783 }
784 }
785 }
786
787 return barriers;
788 }

790 void RenderGraph::simulatePassGroupExecution(
791 const std::string& groupLeader,
792 const std::vector<std::string>& groupMembers,
793 std::unordered_map<Image*, ImageState>& imageStates) {
794
795 for (const std::string& passName : groupMembers) {
796 auto passIt = m_passNodes.find(passName);
797 if (passIt == m_passNodes.end()) continue;
798
799 const auto& passNode = passIt->second;
800
801 for (const auto* targetEdge : passNode.getTargetOutgoingEdges()) {
802 auto imageNodeIt = m_imageNodes.find(targetEdge->toImage);
803 if (imageNodeIt == m_imageNodes.end()) continue;
804
805 Image* image = imageNodeIt->second->getImage();
806 if (!image) continue;
807
808 ImageState& state = imageStates[image];
809 state.currentLayout = ImageLayout::colorAttachmentOptimal;
810 state.lastStage = PipelineStage::colorAttachmentOutput;
811 state.lastAccess = AccessFlag::colorAttachmentWrite;
812 state.lastWriterGroup = groupLeader;
813 }
814
815 for (const auto* depthEdge : passNode.getDepthStencilOutgoingEdges()) {
816 auto imageNodeIt = m_imageNodes.find(depthEdge->toImage);
817 if (imageNodeIt == m_imageNodes.end()) continue;
818
819 Image* image = imageNodeIt->second->getImage();
820 if (!image) continue;
821
822 ImageState& state = imageStates[image];
823 state.currentLayout = ImageLayout::depthAttachmentOptimal;
824 state.lastStage = PipelineStage::lateFragmentTests;
825 state.lastAccess = AccessFlag::depthStencilAttachmentWrite;
826 state.lastWriterGroup = groupLeader;
827 }
828
829 for (const auto* textureEdge : passNode.getTextureIncomingEdges()) {
830 auto imageNodeIt = m_imageNodes.find(textureEdge->fromImage);
831 if (imageNodeIt == m_imageNodes.end()) continue;
832
833 Image* image = imageNodeIt->second->getImage();
834 if (!image) continue;
835
836 ImageState& state = imageStates[image];
837 if (state.currentLayout != ImageLayout::shaderReadOnlyOptimal) {
838 state.currentLayout = ImageLayout::shaderReadOnlyOptimal;
839 state.lastStage = convertShaderStageToPipelineStage(textureEdge->stage);
840 state.lastAccess = AccessFlag::shaderRead;
841 }
842 }
843 }

63 {

65 FT_Int32 load_flags = FT_LOAD_DEFAULT | FT_LOAD_NO_BITMAP;
66 if (FT_HAS_COLOR(m_face))

89 void Freetype_Font::outlineToCommands(FT_Outline *outline, VertexPath* ret)
90 {
91 FT_LayerIterator layer_iterator;
92 FT_UInt layer_index;
93 FT_UInt color_index;
94 FT_Bool have_layers;
95
96 layer_iterator.p = nullptr;
97 have_layers = FT_Get_Color_Glyph_Layer(m_face, m_face->glyph->glyph_index, &layer_index, &color_index, &layer_iterator);
98 ret->beginPath(0);
99 ret->endPath();
100 do
101 {
102 ret->beginPath(1);
103 if (have_layers)
104 {
105 FT_Palette_Data palette_data;
106 FT_Error palette_data_error = FT_Palette_Data_Get(m_face, &palette_data);
107 if (palette_data_error == 0)
108 {
109 FT_Color *palette;
110 FT_Error palette_select_error = FT_Palette_Select(m_face, 0, &palette);
111 if (palette_select_error == 0 && color_index < palette_data.num_palette_entries)
112 {
113 FT_Color color = palette[color_index];
114 ret->setColor(Vec4(color.red / 255.0f, color.green / 255.0f, color.blue / 255.0f, color.alpha / 255.0f));
115 }
116 }
117 FT_Error error = FT_Load_Glyph(m_face, layer_index, FT_LOAD_DEFAULT);
118 if (error)
119 {
120 std::cout << "Error loading glyph layer: " << error << std::endl;
121 continue;
122 }
123 }
124
125 FT_Vector* points = outline->points;
126 unsigned char* tags = outline->tags;
127 unsigned short* contours = outline->contours;
128 int n_points = outline->n_points;
129 int n_contours = outline->n_contours;
130 int start = 0;
131 for (int i = 0; i < n_contours; i++)
132 {
133 int end = contours[i];
134 Vec2 last;
135 Vec2 control;
136 bool hasControl = false;
137
138 for (int j = start; j <= end; j++)
139 {
140 Vec2 point(points[j].x / 64.0f, points[j].y / 64.0f);
141
142 if (j == start)
143 {
144 ret->moveTo(point);
145 last = point;
146 }
147 else
148 {
149 if (FT_CURVE_TAG(tags[j]) == FT_CURVE_TAG_ON)
150 {
151 if (hasControl)
152 {
153 ret->curveTo(control,point);
154 hasControl = false;
155 }
156 else
157 {
158 ret->lineTo(point);
159 }
160 last = point;
161 }
162 else if (FT_CURVE_TAG(tags[j]) == FT_CURVE_TAG_CONIC)
163 {
164 if (hasControl)
165 {
166 Vec2 mid((control.x + point.x) / 2, (control.y + point.y) / 2);
167
168 ret->curveTo(control,mid);
169 last = mid;
170 }
171 control = point;
172 hasControl = true;
173 }
174 else if (FT_CURVE_TAG(tags[j]) == FT_CURVE_TAG_CUBIC)
175 {
176 // todo:从FCTv2 搬过来的，其实已经可以使用三次曲线了，添加三次曲线
177 std::cout << "Warning: Cubic Bezier curves are not supported in this implementation." << std::endl;
178 }
179 }
180 }
181 if (hasControl)
182 {
183 Vec2 start_point(points[start].x / 64.0f, points[start].y / 64.0f);
184 ret->curveTo(control,start_point);
185 }
186 else
187 {
188 Vec2 start_point(points[start].x / 64.0f, points[start].y / 64.0f);
189 ret->lineTo(start_point);
190 }
191
192 start = end + 1;
193 }
194 ret->endPath();
195
196 if (have_layers)
197 {
198 have_layers = FT_Get_Color_Glyph_Layer(m_face, m_face->glyph->glyph_index, &layer_index, &color_index, &layer_iterator);
199 }
200 } while (have_layers);
201 }

369 void Context::addModule()
370 {
371 if constexpr (std::is_same_v<T, ResourceManager>)
372 {
373 FCT_SAFE_NEW(m_resourceManager,ResourceManager,this);
374 }
375 else if constexpr (std::is_same_v<T, RenderGraph>)
376 {
377 FCT_SAFE_NEW(m_renderGraph,RenderGraph,this);
378 }
379 else
380 {
381 ferr << "Unsupported module type: " << typeid(T).name() << std::endl;
382 }
383 }

385 template <typename T>
386 void Context::removeModule()
387 {
388 if constexpr (std::is_same_v<T, ResourceManager>)
389 {
390 FCT_SAFE_DELETE(m_resourceManager);
391 } else
392 {
393 ferr << "Unsupported module type: " << typeid(T).name() << std::endl;
394 }
395
396 }

398 template <typename T>
399 T* Context::getModule()
400 {
401
402 if constexpr (std::is_same_v<T, ResourceManager>)
403 {
404 return m_resourceManager;
405 }
406 else if constexpr (std::is_same_v<T, Device>)
407 {
408 return m_resourceDevice;
409 }
410 else if constexpr (std::is_same_v<T, RenderGraph>)
411 {
412 return m_renderGraph;
413 }
414 else

416 ferr << "try to get undefined context module." << std::endl;

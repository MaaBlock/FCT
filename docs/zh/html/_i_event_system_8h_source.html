132 template<typename Event, typename Func>
133 SubscribeId subscribe(Func&& func) {
134 constexpr auto eventTypeId = getEventTypeId<Event>();
135
136 auto subscribeId = m_nextSubscribeId++;
137
138 auto handler = std::make_unique<EventHandler<Event>>(
139 std::function<void(const Event&)>(std::forward<Func>(func))
140 );
141
142 m_handlers[eventTypeId][subscribeId] = std::move(handler);
143
144 return subscribeId;
145 }

147 template<typename Event>
148 void unsubscribe(SubscribeId subscribeId) {
149 constexpr auto eventTypeId = getEventTypeId<Event>();
150
151 auto it = m_handlers.find(eventTypeId);
152 if (it != m_handlers.end()) {
153 it->second.erase(subscribeId);
154 if (it->second.empty()) {
155 m_handlers.erase(it);
156 }
157 }
158 }

160 template<typename Event>
161 void unsubscribeAll() {
162 constexpr auto eventTypeId = getEventTypeId<Event>();
163 m_handlers.erase(eventTypeId);
164 }
165
166 void unsubscribe(SubscribeId subscribeId) {
167 for (auto& [typeId, handlers] : m_handlers) {
168 auto it = handlers.find(subscribeId);
169 if (it != handlers.end()) {
170 handlers.erase(it);
171 if (handlers.empty()) {
172 m_handlers.erase(typeId);
173 }
174 return;
175 }
176 }
177 }

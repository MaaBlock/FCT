18 Image* Context::loadTexture(const std::string& filename)
19 {
20 auto data = m_imageLoader->load(filename);
21 SingleBufferImage* image = new SingleBufferImage(this);
22 image->width(data.width);
23 image->height(data.height);
24
25 bool isSRGB = true;
26
27
28 Format format;
29 switch (data.channels) {
30 case 1:
31 format = Format::R8_UNORM;
32 break;
33 case 2:
34 format = Format::R8G8_UNORM;
35 break;
36 case 3:
37 format = isSRGB ? Format::R8G8B8A8_SRGB : Format::R8G8B8_UNORM;
38 break;
39 case 4:
40 format = isSRGB ? Format::R8G8B8A8_SRGB : Format::R8G8B8A8_UNORM;
41 break;
42 default:
43 ferr << "Unsupported number of channels: " << data.channels << std::endl;
44 format = isSRGB ? Format::R8G8B8A8_SRGB : Format::R8G8B8A8_UNORM;
45 break;
46 }
47
48 image->format(format);
49 image->as(ImageUsage::Texture);
50 image->initData(data.data.data(), data.data.size());
51 image->create();
52 return image;
53 }

170 StaticMesh<uint32_t>* Context::createMesh(const ModelMesh* modelMesh, const VertexLayout& layout)
171 {
172 if (!modelMesh) {
173 return nullptr;
174 }
175
176 StaticMesh<uint32_t>* mesh = new StaticMesh<uint32_t>(this, layout);
177 VertexBuffer* vertexBuffer = mesh->getVertexBuffer();
178 vertexBuffer->resize(modelMesh->vertices.size());
179 for (uint32_t i = 0; i < modelMesh->vertices.size(); ++i) {
180 const auto& modelVertex = modelMesh->vertices[i];
181 Vertex vertex = (*vertexBuffer)[i];
182
183 for (size_t j = 0; j < layout.getElementCount(); ++j) {
184 const VertexElement& element = layout.getElement(j);
185 setVertexAttributeFromModel(vertex, j, element, modelVertex);
186 }
187 }
188
189 std::vector<uint32_t> indices = modelMesh->indices;
190 mesh->setIndices(indices);
191
192 mesh->create();
193
194 return mesh;
195 }

241 void update()
242 {
243 std::deque<BoostVertex> sortedVertices;
244
245 std::map<BoostVertex, std::size_t> vertex_index_map;
246 std::size_t index = 0;
247 for (auto [v_iter, v_end] = boost::vertices(m_graph); v_iter != v_end; ++v_iter) {
248 vertex_index_map[*v_iter] = index++;
249 }
250 auto index_pmap = boost::make_assoc_property_map(vertex_index_map);
251
252 std::map<BoostVertex, boost::default_color_type> vertex_color_map;
253 auto color_pmap = boost::make_assoc_property_map(vertex_color_map);
254
255 boost::topological_sort(m_graph, std::front_inserter(sortedVertices),
256 boost::vertex_index_map(index_pmap).color_map(color_pmap));
257
258 m_order.clear();
259
260 for (auto vertex : sortedVertices) {
261 auto it = m_vertex.right.find(vertex);
262 if (it != m_vertex.right.end()) {
263 Token token = it->second;
264 if (hasNode(token)) {
265 m_order.push_back(m_nodeMap[token].value);
266 }
267 }

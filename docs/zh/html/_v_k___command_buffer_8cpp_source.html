55 {
56 vk::SubmitInfo submitInfo{};
57 submitInfo.setCommandBuffers(m_commandBuffer);
58 std::vector<vk::Semaphore> waitSemaphores;
59 std::vector<vk::PipelineStageFlags> waitStages;
60 for (auto& desc : m_waitSemaphores)
61 {
62 waitSemaphores.push_back(static_cast<VK_Semaphore*>(desc.semaphore)->semaphore());
63 waitStages.push_back(static_cast<vk::PipelineStageFlags>(static_cast<size_t>(desc.stages)));
64 }
65 submitInfo.setWaitSemaphores(waitSemaphores);
66 submitInfo.setWaitDstStageMask(waitStages);
67 std::vector<vk::Semaphore> signalSemaphores;
68 for (auto& signalSemaphore : m_signalSemaphores)
69 {
70 signalSemaphores.push_back(static_cast<VK_Semaphore*>(signalSemaphore)->semaphore());
71 }
72 submitInfo.setSignalSemaphores(signalSemaphores);
73 m_pool->context()->getGraphicsQueue().submit(submitInfo,
74 static_cast<VK_Fence*>(m_fence)->fence());
